Dead Simple Integration

Deploy your WebAssembly functions into the swarm with just a few lines of code.

import { SwarmMesh } from '@swarmcompute/client';

// Initialize the P2P connection to the swarm router
const mesh = new SwarmMesh({
  apiKey: 'sc_live_xxxxx',
  maxThreads: navigator.hardwareConcurrency || 4,
  memoryLimit: '512MB'
});

await mesh.connect();
console.log('Node joined the swarm network.');
#[wasm_bindgen]
pub fn process_image_chunk(data: &[u8], filter: &str) -> Vec<u8> {
    // Heavy computation running in browser sandbox
    let mut image = image::load_from_memory(data).unwrap();
    
    if filter == "grayscale" {
        image = image.grayscale();
    }
    
    let mut output = Vec::new();
    image.write_to(&mut output, image::ImageOutputFormat::Png).unwrap();
    
    output
}
// Dispatch workload to the mesh network
const imageBuffer = await fetch('/large-image.raw').then(r => r.arrayBuffer());

// Automatically chunks and distributes across available nodes
const result = await mesh.dispatch('process_image_chunk', {
  payload: imageBuffer,
  args: ['grayscale'],
  distributeStrategy: 'scatter-gather'
});

renderImage(result.assembledPayload);
